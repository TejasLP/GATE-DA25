\begin{center}
	\begin{tikzpicture}[
		>=Stealth,
		vertex/.style={circle, draw=black, very thick, minimum size=8.5mm, inner sep=0pt, font=\bfseries},
		discover_even/.style={->, very thick, black},
		discover_odd/.style={->, thick, dashed, black},
		backtrack/.style={->, thick, dotted, black},
		graphedge/.style={->, thin, loosely dotted, black!45}
		]
		
		% Top row (Even vertices: Forward chain)
		\node[vertex] (v0) at (0, 0) {0};
		\node[vertex] (v2) at (3.2, 0) {2};
		\node[vertex] (v4) at (6.4, 0) {4};
		\node[vertex] (v6) at (9.6, 0) {6};
		
		% Bottom row (Odd vertices: Branch discoveries)
		\node[vertex] (v1) at (1.6, -2.2) {1};
		\node[vertex] (v3) at (4.8, -2.2) {3};
		\node[vertex] (v5) at (8.0, -2.2) {5};
		
		% Remaining edges of the graph G (Unvisited)
		\draw[graphedge] (v1) -- (v2);
		\draw[graphedge] (v1) -- (v3);
		\draw[graphedge] (v3) -- (v4);
		\draw[graphedge] (v3) -- (v5);
		\draw[graphedge] (v5) -- (v6);
		
		% ==========================================
		% Phase 1: Forward greedy path on even nodes
		% ==========================================
		\draw[discover_even] (v0) -- node[above=2pt, font=\small\bfseries] {Step 1} (v2);
		\draw[discover_even] (v2) -- node[above=2pt, font=\small\bfseries] {Step 2} (v4);
		\draw[discover_even] (v4) -- node[above=2pt, font=\small\bfseries] {Step 3} (v6);
		
		% Dead end at node 6
		\node[above=2pt of v6, font=\scriptsize\itshape] {Dead end};
		
		% ==========================================
		% Backtrack 1: From 6 back to 4
		% ==========================================
		\draw[backtrack] (v6) to[bend right=35] node[above=1pt, font=\scriptsize\itshape] {Backtrack to 4} (v4);
		
		% Step 4: Explore vertex 5 from vertex 4
		\draw[discover_odd] (v4) -- node[pos=0.45, right=2pt, font=\small\bfseries] {Step 4} (v5);
		\node[below=2pt of v5, font=\scriptsize\itshape] {Dead end};
		
		% ==========================================
		% Backtrack 2: Unwind call stack (5 -> 4 -> 2)
		% ==========================================
		\draw[backtrack] (v5) to[bend left=25] node[left=1pt, font=\tiny\itshape] {return} (v4);
		\draw[backtrack] (v4) to[bend right=35] node[above=1pt, font=\scriptsize\itshape] {Backtrack to 2} (v2);
		
		% Step 5: Explore vertex 3 from vertex 2
		\draw[discover_odd] (v2) -- node[pos=0.45, right=2pt, font=\small\bfseries] {Step 5} (v3);
		\node[below=2pt of v3, font=\scriptsize\itshape] {Dead end};
		
		% ==========================================
		% Backtrack 3: Unwind call stack (3 -> 2 -> 0)
		% ==========================================
		\draw[backtrack] (v3) to[bend left=25] node[left=1pt, font=\tiny\itshape] {return} (v2);
		\draw[backtrack] (v2) to[bend right=35] node[above=1pt, font=\scriptsize\itshape] {Backtrack to 0} (v0);
		
		% Step 6: Explore vertex 1 from vertex 0
		\draw[discover_odd] (v0) -- node[pos=0.45, right=2pt, font=\small\bfseries] {Step 6} (v1);
		\node[below=2pt of v1, font=\scriptsize\itshape] {Complete};
		
	\end{tikzpicture}
\end{center}
